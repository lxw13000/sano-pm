
SELECT
  base.order_no,
  base.trade_no,
  base.producer_type,
  base.producer_code,
  base.channel_code,
  base.user_id,
  base.diamond,
  base.diamond_old,
  base.diamond_new,
  base.money_dollar,
  base.reality_money,
  base.service_fixed_money,
  base.service_float_rate,
  base.service_float_money,
  base.user_withdraw_channel_id,
  base.currency_icon,
  base.currency_code,
  base.rate,
  base.money,
  base.STATUS,
  base.country_code,
  base.country_flag_square,
  base.country_flag_round,
  base.create_time,
  base.pay_time,
  base.reject_reason,
  base.audit_user,
  base.audit_time,
  base.remarks,
  u.user_code AS userCode,
  wc.account_no AS accountNo,
  wc.full_name AS fullName,
  wc.payee_phone AS payeePhone,
  wc.usdt_address AS usdtAddress,
  wc.binance_id AS binanceId,
  wc.email AS email,
  wc.account_type AS accountType,
  wc.check_digit AS checkDigit,
  wc.bank_code AS bankCode,
  wc.bank_name AS bankName,
  wc.bank_branch AS bankBranch,
  wc.bank_city AS bankCity,
  wc.cor_account_no AS corAccountNo,
  wc.first_name AS firstName,
  wc.middle_name AS middleName,
  wc.last_name AS lastName,
  wc.document_type AS documentType,
  wc.document_id AS documentId,
  wc.documentIssue_date AS documentIssueDate,
  wc.document_expire_date AS documentExpireDate,
  wc.address AS address,
  wc.city AS city,
  wc.`state` AS state,
  wc.zip_code AS zipCode,
  wc.birth_date AS birthDate
FROM
  sano_wallet_withdraw base
  LEFT JOIN sano_user u ON base.user_id = u.id
  LEFT JOIN sano_user_withdraw_channel wc ON base.user_withdraw_channel_id = wc.id
