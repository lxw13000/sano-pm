CREATE TABLE `sano_log_device` (
  `id` bigint NOT NULL AUTO_INCREMENT COMMENT '记录ID，服务端数据库生成',
  `user_id` int DEFAULT NULL COMMENT '通过Token解析的当前用户ID',
  `create_time` datetime(3) NOT NULL COMMENT '服务端接收时间，权威时间及日期归属依据',
  `app_install_id` varchar(128) CHARACTER SET utf8mb4 COLLATE utf8mb4_bin DEFAULT NULL COMMENT '安装实例ID，生成或读取失败允许为空',
  `brand_code` varchar(64) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci DEFAULT NULL COMMENT '设备品牌或制造商',
  `device_model` varchar(128) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci DEFAULT NULL COMMENT '系统原始设备型号代码',
  `device_time` bigint DEFAULT NULL COMMENT '设备系统当前时间，Unix毫秒时间戳',
  `mac` varchar(64) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci DEFAULT NULL COMMENT 'HttpRequestVO的mac，沿用adId/deviceId设备标识口径，可空',
  `os_version` varchar(128) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci DEFAULT NULL COMMENT '操作系统及版本，如Android 15、iOS 18.0',
  `app_issue_version` varchar(64) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci DEFAULT NULL COMMENT 'App发布版本号',
  `app_build_version` varchar(64) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci DEFAULT NULL COMMENT 'App内部构建号，按字符串保留原值',
  `network_type` varchar(16) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci DEFAULT NULL COMMENT '网络类型原值，非标准值也保留，可空',
  `rs` tinyint DEFAULT NULL COMMENT '是否模拟器：1是、0否、NULL未检测或无法确定',
  `ra` tinyint DEFAULT NULL COMMENT '是否开启VPN或代理：1是、0否、NULL未检测或无法确定',
  `ip` varchar(45) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci DEFAULT NULL COMMENT 'HttpRequestVO的IP，沿用现有拦截器取值，兼容IPv4和IPv6',
  `ip_country_code` varchar(16) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci DEFAULT NULL COMMENT 'IP国家编码，解析失败为空',
  `ip_region` varchar(128) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci DEFAULT NULL COMMENT 'IP地区或省州，库不支持或解析失败为空',
  `ip_city` varchar(128) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci DEFAULT NULL COMMENT 'IP城市，库不支持或解析失败为空',
  `register_country_code` varchar(16) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci DEFAULT NULL COMMENT '用户注册国家编码，服务端取得',
  `register_channel_code` varchar(255) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci DEFAULT NULL COMMENT '注册渠道，取sano_user_register.ad_channel，缺失为空',
  `guild_owner_flag` tinyint DEFAULT NULL COMMENT '是否审核通过且未删除的公会长：1是、0否、NULL查询失败',
  `payroll_flag` tinyint DEFAULT NULL COMMENT '是否有效Payroll代理：level大于0且未删除，1是、0否、NULL查询失败',
  `bd_flag` tinyint DEFAULT NULL COMMENT '是否BD：存在sys_user.bd_user_id关联，1是、0否、NULL查询失败',
  `client_type` varchar(20) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci DEFAULT NULL COMMENT '客户端类型EClientType',
  PRIMARY KEY (`id`) USING BTREE,
  KEY `idx_create_time` (`create_time`) USING BTREE,
  KEY `idx_user_time` (`user_id`,`create_time`) USING BTREE,
  KEY `idx_install_time` (`app_install_id`,`create_time`) USING BTREE,
  KEY `idx_ip_time` (`ip`,`create_time`) USING BTREE
) ENGINE=InnoDB DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci;
