CREATE TABLE `sano_room` (
  `id` int NOT NULL AUTO_INCREMENT COMMENT '房间ID',
  `user_id` int DEFAULT NULL COMMENT '房主ID',
  `name` varchar(32) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci DEFAULT NULL COMMENT '房间名称',
  `notice` varchar(256) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci DEFAULT NULL COMMENT '公告',
  `country_code` varchar(10) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci DEFAULT NULL COMMENT '国家代码',
  `country_icon` varchar(255) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci DEFAULT NULL COMMENT '国旗',
  `cover` varchar(64) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci DEFAULT NULL COMMENT '封面图片',
  `background` varchar(64) CHARACTER SET utf8mb4 COLLATE utf8mb4_0900_ai_ci DEFAULT NULL COMMENT '背景图片',
  `room_type` tinyint DEFAULT NULL COMMENT '1 单人直播，2 多人语音，3 多人直播',
  `room_classify` tinyint DEFAULT NULL COMMENT '房间类型：1 单人直播，2 多人语音，3 多人直播，4 9人间',
  `sound_tag` int DEFAULT NULL COMMENT '语音房标签： 1 聊天，2 交友，3 跳舞，4 游戏，5 搞笑',
  `mai_amount` tinyint DEFAULT '5' COMMENT '麦位数量',
  `new_user_status` tinyint(1) DEFAULT '0' COMMENT '是否是新用户',
  `online_count` bigint DEFAULT '0' COMMENT '累积在线人数',
  `online_count_now` bigint DEFAULT NULL COMMENT '当前在线人数',
  `shang_mai_count` int DEFAULT '0' COMMENT '累积上麦次数',
  `open_mai` tinyint DEFAULT '1' COMMENT '是否开启连麦',
  `open_auto_mai` tinyint DEFAULT '0' COMMENT '是否开启自动上麦',
  `order_index` int DEFAULT '0' COMMENT '排序权重，大到小',
  `create_time` datetime DEFAULT CURRENT_TIMESTAMP COMMENT '开播时间',
  `pk_id` bigint DEFAULT '0' COMMENT '房间PKID，0 未PK',
  `red_id` bigint DEFAULT '0' COMMENT '红包ID，0 没有红包',
  `status` tinyint DEFAULT '0' COMMENT '是否关播',
  `coin` bigint DEFAULT '0' COMMENT '累积普通金币',
  `diamond` decimal(64,3) DEFAULT '0.000' COMMENT '累积普通钻石',
  `coin_lucky` bigint DEFAULT NULL COMMENT '幸运金币',
  `diamond_lucky` decimal(64,3) DEFAULT NULL COMMENT '幸运钻石',
  `live_second` int DEFAULT '0' COMMENT '直播时长秒',
  `close_time` datetime DEFAULT NULL COMMENT '关播时间',
  `camera_reverse` tinyint(1) DEFAULT NULL COMMENT '摄像头是否反转',
  `heat_value` bigint DEFAULT '0' COMMENT '热度值',
  `float_heat_value` bigint DEFAULT '0' COMMENT '浮动热度值',
  `language_code` varchar(32) DEFAULT NULL COMMENT '房主语言编码',
  `quality_anchor` tinyint DEFAULT '0' COMMENT '是否优质主播',
  `rtc_type` tinyint DEFAULT '1' COMMENT '流类型：1 机构，2 腾讯',
  PRIMARY KEY (`id`) USING BTREE,
  KEY `userId` (`user_id`) USING BTREE,
  KEY `createTime` (`create_time`) USING BTREE,
  KEY `statusTime` (`status`,`create_time` DESC) USING BTREE,
  KEY `idx_cover_user` (`user_id`,`status`,`close_time`,`room_type`,`create_time`) USING BTREE,
  KEY `idx_cover_all` (`status`,`close_time`,`user_id`,`room_type`,`create_time`) USING BTREE,
  KEY `idx_user_create_close_type` (`user_id`,`create_time`,`close_time`,`room_type`) USING BTREE
) ENGINE=InnoDB AUTO_INCREMENT=5032204 DEFAULT CHARSET=utf8mb4 COLLATE=utf8mb4_0900_ai_ci COMMENT='直播间';
